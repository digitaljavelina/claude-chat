"""End-to-end canary test: full pipeline on synthetic PII → zero CANARY survivors.

Run with: python3 -m unittest tests.test_scrub_canary -v

This is the PRIV-04 acceptance test. It exercises:
  load (JSONL) → scrub (scrub_content in _get_markdown_body) →
  label (stub) → write (cmd_write) → grep vault bytes

Mocks claude-chat.py via SYNC_CHATS_CLAUDE_CLI so the test doesn't need a real
claude-chat.py install — we just stream the canary JSONL content back as if
claude-chat.py export had rendered it.

Asserts:
  - No CANARY substring in vault file
  - Private IPs (192.168.1.100, 127.0.0.1) preserved
  - Redaction tokens present
  - sync.log contains no matched substrings (PRIV-06)
"""

import io
import json
import os
import shutil
import socket
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path
from unittest import mock

_GLOBAL_TEMP = tempfile.mkdtemp()
os.environ["CLAUDE_CHAT_HOME"] = _GLOBAL_TEMP

sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))
import sync_chats  # noqa: E402


CANARY_JSONL = os.path.join(os.path.dirname(__file__), "canary_session.jsonl")


def _canary_body_text() -> str:
    """Extract the user message content from canary_session.jsonl.

    This is what a real `claude-chat.py export --format md --stdout` would
    produce (roughly — we skip the markdown wrappers and just use the content
    for scrub testing, since scrub runs on the body as a str regardless).
    """
    parts = []
    with open(CANARY_JSONL, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            obj = json.loads(line)
            msg = obj.get("message", {})
            content = msg.get("content", "")
            if content:
                role = msg.get("role", "?")
                parts.append(f"## {role}\n\n{content}\n")
    return "\n".join(parts)


class TestCanaryPipeline(unittest.TestCase):
    """PRIV-04: feed canary through full pipeline, grep vault for CANARY — expect zero."""

    def setUp(self):
        # Isolate everything: home + vault + projects in tempdirs
        self.tmp_home = Path(tempfile.mkdtemp())
        self.tmp_vault = Path(tempfile.mkdtemp())
        self.tmp_projects = Path(tempfile.mkdtemp())
        self.addCleanup(lambda: shutil.rmtree(self.tmp_home, ignore_errors=True))
        self.addCleanup(lambda: shutil.rmtree(self.tmp_vault, ignore_errors=True))
        self.addCleanup(lambda: shutil.rmtree(self.tmp_projects, ignore_errors=True))

        # Save and override sync_chats module paths
        self._saved = (
            sync_chats.CLAUDE_CHAT_HOME, sync_chats.CONFIG_PATH,
            sync_chats.STATE_PATH, sync_chats.LOG_PATH, sync_chats.PROJECTS_DIR,
        )
        sync_chats.CLAUDE_CHAT_HOME = self.tmp_home
        sync_chats.CONFIG_PATH = self.tmp_home / "config.json"
        sync_chats.STATE_PATH = self.tmp_home / "state.json"
        sync_chats.LOG_PATH = self.tmp_home / "sync.log"
        sync_chats.PROJECTS_DIR = self.tmp_projects

        # Seed config + state
        sync_chats.CONFIG_PATH.write_text(json.dumps({
            "schema_version": 1,
            "machine_label": "canary",
            "vault_path": str(self.tmp_vault),
        }))
        sync_chats.STATE_PATH.write_text(json.dumps({
            "schema_version": 1,
            "synced_session_ids": [],
            "fingerprints": {},
            "last_run_at": None,
        }))

        # Place a synthetic JSONL at the right depth: PROJECTS_DIR/<proj>/<sid>.jsonl
        self.session_id = "canary00-0000-0000-0000-000000000000"
        proj_dir = self.tmp_projects / "canary-project"
        proj_dir.mkdir()
        shutil.copy(CANARY_JSONL, proj_dir / f"{self.session_id}.jsonl")

    def tearDown(self):
        (sync_chats.CLAUDE_CHAT_HOME, sync_chats.CONFIG_PATH,
         sync_chats.STATE_PATH, sync_chats.LOG_PATH,
         sync_chats.PROJECTS_DIR) = self._saved

    def test_full_pipeline_zero_canary_survivors(self):
        """ROADMAP SC#1 + PRIV-04: grep vault for CANARY → zero hits."""
        canary_body = _canary_body_text()

        # Patch _get_markdown_body's subprocess call to return the canary body
        # as if claude-chat.py export had rendered it. The scrub step inside
        # _get_markdown_body will still run normally (that's what we're testing).
        fake_result = subprocess.CompletedProcess(
            args=[], returncode=0, stdout=canary_body, stderr=""
        )

        # Build a stub label JSON and feed it via stdin
        label_json = json.dumps({
            "title": "Canary test session",
            "gist": None,
            "tags": ["canary", "test"],
            "coherence_score": None,
            "needs_review": False,  # will be forced to True if uncertain fires
        })

        # Mock subprocess.run for _get_markdown_body and sys.stdin for cmd_write
        args = mock.MagicMock()
        args.session_id = self.session_id

        with mock.patch("sync_chats.subprocess.run", return_value=fake_result), \
             mock.patch("sys.stdin", io.StringIO(label_json)):
            try:
                sync_chats.cmd_write(args)
            except SystemExit as e:
                # Should exit 0 on success. cmd_write may not raise SystemExit in the
                # success path — it just returns. Only fail if exit code is non-zero.
                if e.code not in (0, None):
                    raise

        # Find the written vault file (canary--2026-04-13--canary-test-session.md or similar)
        chats_dir = self.tmp_vault / "Chats"
        vault_files = list(chats_dir.glob("*.md"))
        self.assertEqual(len(vault_files), 1, f"expected 1 vault file, got {len(vault_files)}")
        vault_file = vault_files[0]
        vault_bytes = vault_file.read_bytes()

        # ─── PRIV-04 / ROADMAP SC#1: zero CANARY substrings survive ───
        self.assertNotIn(b"CANARY", vault_bytes,
                         f"CANARY substring survived scrub in {vault_file.name}")

        # Sanity: redaction actually happened (not just scrubbed by deletion)
        self.assertIn(b"<REDACTED:", vault_bytes,
                      "no <REDACTED:...> tokens in vault file — did scrub run?")

        # ─── D-10 skip-list: private IPs preserved ───
        self.assertIn(b"192.168.1.100", vault_bytes,
                      "private IP 192.168.1.100 was wrongly redacted")
        self.assertIn(b"127.0.0.1", vault_bytes,
                      "loopback 127.0.0.1 was wrongly redacted")

        # ─── D-08: privacy_review frontmatter field present ───
        head = vault_file.read_text()[:2000]
        self.assertTrue(
            "privacy_review: scrubbed" in head or "privacy_review: uncertain" in head,
            f"privacy_review not in scrubbed|uncertain state — frontmatter head:\n{head[:400]}"
        )

    def test_sync_log_contains_no_canary(self):
        """PRIV-06 + ROADMAP SC#4: sync.log has pattern names + counts only."""
        canary_body = _canary_body_text()
        fake_result = subprocess.CompletedProcess(
            args=[], returncode=0, stdout=canary_body, stderr=""
        )
        label_json = json.dumps({
            "title": "Canary log test",
            "tags": ["canary"],
            "needs_review": False,
        })
        args = mock.MagicMock()
        args.session_id = self.session_id

        with mock.patch("sync_chats.subprocess.run", return_value=fake_result), \
             mock.patch("sys.stdin", io.StringIO(label_json)):
            try:
                sync_chats.cmd_write(args)
            except SystemExit as e:
                if e.code not in (0, None):
                    raise

        log_text = sync_chats.LOG_PATH.read_text() if sync_chats.LOG_PATH.exists() else ""
        # PRIV-06: log must not contain the matched substrings
        self.assertNotIn("CANARY", log_text,
                         f"CANARY substring leaked into sync.log: {log_text}")
        # D-21 format: the scrub line should contain "scrub session=" + "patterns={"
        self.assertIn("scrub session=", log_text,
                      "expected a D-21 scrub log line in sync.log")
        self.assertIn("patterns={", log_text)
        # Spot-check: no @ sign (would indicate email leaked)
        # (Note: timestamps use `:` not `@`, so `@` is a reliable email tell.)
        self.assertNotIn("@", log_text, "@ sign in sync.log suggests email leaked")


if __name__ == "__main__":
    unittest.main()
